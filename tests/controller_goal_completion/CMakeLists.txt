cmake_minimum_required(VERSION 3.16)
project(axisforge_controller_goal_completion LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(control_msgs REQUIRED)
add_executable(goal_completion goal_completion.cpp)
target_compile_features(goal_completion PRIVATE cxx_std_17)
if(PATCHED_INCLUDE)
  target_include_directories(goal_completion BEFORE PRIVATE "${PATCHED_INCLUDE}")
endif()
ament_target_dependencies(goal_completion rclcpp rclcpp_action realtime_tools control_msgs)
if(CONTROLLER_LIBRARY)
  target_compile_definitions(goal_completion PRIVATE TEST_LIBRARY_CALLBACK)
  target_link_libraries(goal_completion "${CONTROLLER_LIBRARY}")
endif()
